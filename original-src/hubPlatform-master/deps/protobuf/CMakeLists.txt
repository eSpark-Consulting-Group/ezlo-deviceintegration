cmake_minimum_required( VERSION 3.2 )
include(ExternalProject)

set(PROTOBUF_EXT_PROJECT protobuf_ext)
set(PROTOBUF_INSTALL_DIR ${CMAKE_CURRENT_BINARY_DIR})

list(APPEND CMAKE_OPTS
    -DCMAKE_C_COMPILER=${CMAKE_C_COMPILER}
    -DCMAKE_CXX_COMPILER=${CMAKE_CXX_COMPILER}
    -DCMAKE_CXX_FLAGS=${CMAKE_CXX_FLAGS}
    -DCMAKE_CXX_STANDARD=${CMAKE_CXX_STANDARD}
    -DCMAKE_AR=${CMAKE_AR}
    -DCMAKE_LINKER=${CMAKE_LINKER}
    -DCMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}
    -DCMAKE_INSTALL_PREFIX=${PROTOBUF_INSTALL_DIR}
)

list(APPEND PROTOBUF_OPTS
    -DBUILD_SHARED_LIBS=OFF
    -DBUILD_STATIC_LIBS=ON
    -Dprotobuf_BUILD_TESTS=OFF
)

ExternalProject_Add(
    ${PROTOBUF_EXT_PROJECT}
    CMAKE_ARGS ${CMAKE_OPTS} ${PROTOBUF_OPTS}
    URL ${CMAKE_CURRENT_SOURCE_DIR}/protobuf-cpp-3.21.9.tar.gz

    LOG_DOWNLOAD 1
    LOG_BUILD 1
    LOG_INSTALL 1
)
