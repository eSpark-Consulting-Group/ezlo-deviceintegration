cmake_minimum_required( VERSION 3.2 )
include(ExternalProject)

set(PACKAGE_NAME firmware-common-mqtt_client)

set(PAHO_MQTT_INSTALL_DIR ${CMAKE_CURRENT_BINARY_DIR})

list(APPEND CMAKE_OPTS
     -DCMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}
     -DCMAKE_INSTALL_PREFIX=${PAHO_MQTT_INSTALL_DIR}
     )

list(APPEND PAHO_MQTT_C_OPTS
     -DCMAKE_C_COMPILER=${CMAKE_C_COMPILER}
     -DPAHO_ENABLE_TESTING=OFF
     -DPAHO_WITH_SSL=FALSE
     -DPAHO_HIGH_PERFORMANCE=ON
     -DPAHO_BUILD_DOCUMENTATION=FALSE
     -DPAHO_BUILD_SAMPLES=FALSE
     -DPAHO_ENABLE_CPACK=FALSE
     )

list(APPEND PAHO_MQTT_CPP_OPTS
     -DCMAKE_CXX_COMPILER=${CMAKE_CXX_COMPILER}
     -DPAHO_BUILD_DOCUMENTATION=FALSE
     -DPAHO_BUILD_SAMPLES=FALSE
     -DPAHO_WITH_SSL=OFF
     )

set(PAHO_MQTT_C_PROJECT pahomqttc)
set(PAHO_MQTT_CPP_PROJECT pahomqttcpp)

ExternalProject_Add(
    ${PAHO_MQTT_C_PROJECT}
    CMAKE_ARGS ${CMAKE_OPTS} ${PAHO_MQTT_C_OPTS}
    URL ${CMAKE_CURRENT_SOURCE_DIR}/paho.mqtt.c-1.3.8.zip
    LOG_DOWNLOAD 1
    )

ExternalProject_Add(
    ${PAHO_MQTT_CPP_PROJECT}
    DEPENDS ${PAHO_MQTT_C_PROJECT}
    CMAKE_ARGS ${CMAKE_OPTS} ${PAHO_MQTT_CPP_OPTS}
    URL ${CMAKE_CURRENT_SOURCE_DIR}/paho.mqtt.cpp-1.2.0.zip
    LOG_DOWNLOAD 1
    )

ExternalProject_Add_Step(
    ${PAHO_MQTT_CPP_PROJECT} copy2bin
    COMMAND ${CMAKE_COMMAND} -E copy_directory ${PAHO_MQTT_INSTALL_DIR}/lib ${CMAKE_BINARY_DIR}/lib
    ALWAYS 1
    DEPENDEES install
)

install(DIRECTORY ${PAHO_MQTT_INSTALL_DIR}/lib DESTINATION ${PACKAGE_NAME}/opt/firmware FILES_MATCHING PATTERN "libpaho*" PATTERN "cmake" EXCLUDE )
